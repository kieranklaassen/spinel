===== 3. the per-kind generator (gen_kinds.rb) =====
# gen.rb: one program per kind of receiver of a method added to a builtin, beside an argument
# that runs the program's code: `recv.pair(arg)`, both logging. h_*: recv_held_as_read holds the
# receiver first; m_*: a String another name may hold, or a boxed value, keeps master's C.
K = {
  "h_int"    => ["Integer", "5"],
  "h_flt"    => ["Float", "2.5"],
  "h_sym"    => ["Symbol", ":s"],
  "h_bool"   => ["TrueClass", "true"],
  "h_rng"    => ["Range", "(1..3)"],
  "h_ary"    => ["Array", "[1, 2]"],
  "h_hsh"    => ["Hash", "{ 1 => 2 }"],
  "h_obj"    => ["Object", "Foo.new"],
  "h_interp" => ["String", "\"a\#{ARGV.size}\""],
  "h_built"  => ["String", "$s.upcase"],
  "h_tos"    => ["String", "ARGV.size.to_s"],
  "h_meth"   => ["String", "fresh"],
  "m_glob"   => ["String", "$s"],
  "m_param"  => ["String", "same($s)"],
  "m_ivar"   => ["String", "$box.s"],
  "m_ret"    => ["String", "early"],
}
K.each do |n, (cls, val)|
  File.write("#{n}.rb", <<~RB)
    $log = []
    $s = +"ab"
    class Foo; end
    class Box
      attr_reader :s
      def initialize = @s = $s
    end
    $box = Box.new
    def fresh = "f\#{ARGV.size}"
    def same(x) = x
    def early
      return $s if ARGV.size == 0
      "e\#{ARGV.size}"
    end
    class #{cls}
      def pair(a) = "\#{self.class}|\#{self.size rescue 0}|\#{a}"
    end
    def recv = ($log << "recv"; #{val})
    def arg = ($log << "arg"; $s << "c"; 1)
    puts recv.pair(arg)
    p $log
  RB
end
